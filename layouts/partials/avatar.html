{{- /* Accepts a name string (renders initials) OR a dict {name, image} (renders a headshot). */ -}} {{- $name := . -}} {{- $image := "" -}} {{- if reflect.IsMap . -}} {{- $name = .name -}} {{- $image = .image -}} {{- end -}} {{- if $image -}} {{- else -}} {{- /* Skip honorifics so "Dr Osa Edosa" gives "OE", not "DO". */ -}} {{- $titles := slice "dr" "dr." "prof" "prof." "engr" "engr." "mr" "mr." "mrs" "mrs." "ms" "ms." "miss" -}} {{- $parts := slice -}} {{- range split $name " " -}} {{- if and . (not (in $titles (lower .))) -}}{{- $parts = $parts | append . -}}{{- end -}} {{- end -}} {{- $initials := "" -}} {{- range first 2 $parts -}} {{- $initials = printf "%s%s" $initials (substr . 0 1) -}} {{- end -}}
{{ $initials | upper }}
{{- end -}}
